{{- /* apps.brianpfeil.com: a directory of every app in data/apps.json, one card each (icon, name, blurb), every card opening its app in a new tab. One self-contained file, /apps-home/index.html: CSS inline and every icon a data: URI, so it needs nothing else from this site. A CloudFront distribution (infra/apps.tf) serves it as apps.brianpfeil.com/ with origin path /apps-home; its viewer function sends apps.brianpfeil.com/ to the /apps// short link and anything else back to /. On brianpfeil.com itself the file is noindex with apps.brianpfeil.com as its canonical, so search engines see one page. English, like the short links: called once, from the English home page. Strings are the /apps/ page's own (apps_intro, apps_blurb_, …). */ -}} {{- $apps := hugo.Data.apps.apps -}} {{- $favicon := printf "data:image/svg+xml;base64,%s" (readFile "static/favicon.svg" | base64Encode) -}} {{- $cards := slice -}} {{- range $apps -}} {{- $tags := slice -}} {{- if .private }}{{ $tags = $tags | append (i18n "apps_private") }}{{ end -}} {{- if .sign_in }}{{ $tags = $tags | append (i18n "apps_sign_in") }}{{ end -}} {{- if .offline }}{{ $tags = $tags | append (i18n "apps_offline") }}{{ end -}} {{- $icon := printf "data:image/webp;base64,%s" (readFile (printf "static%s" .icon) | base64Encode) -}} {{- $host := replaceRE `^https://([^/]+).*$` "$1" .url -}} {{- $card := printf `
  • %s↗ %s %s
    • ` (.url | htmlEscape) .key $icon (.name | htmlEscape) (i18n (printf "apps_blurb_%s" .key) | htmlEscape) (delimit (slice $host | append $tags) " · " | htmlEscape) -}} {{- $cards = $cards | append $card -}} {{- end -}} {{- $title := i18n "hub_apps_title" -}} {{- $intro := i18n "apps_portal_intro" -}} {{- $html := printf ` %s · Brian Pfeil

    %s

    %s

      %s
    brianpfeil.com · brianpfeil.com/apps
    ` ($title | htmlEscape) ($intro | htmlEscape) $favicon ($title | htmlEscape) ($intro | htmlEscape) (delimit $cards "\n") -}} {{- with resources.FromString "apps-home/index.html" $html -}} {{- .Publish -}} {{- end -}}
