{{- /* Every file in data/ is also published as /data/.json, and the data-driven pages (/music/, /subscriptions/, /media/, /movies/) are drawn in the browser from those copies. Hugo reads YAML and JSON alike, so this covers both. Called once, from the English home page; resources publish to the site root either way. media and photos are the exceptions: they get the whitelisted payloads from partials/media-payload.html and photos-payload.html, never the raw files. */ -}} {{- range $name, $d := hugo.Data -}} {{- $out := $d -}} {{- if eq $name "media" }}{{ $out = partial "media-payload.html" . }}{{ end -}} {{- if eq $name "photos" }}{{ $out = partial "photos-payload.html" . }}{{ end -}} {{- with resources.FromString (printf "data/%s.json" $name) (jsonify $out) -}} {{- .Publish -}} {{- end -}} {{- /* movies.json carries titles, genres and synopses in all nine languages (544 KB); a visitor reads one. /movies/ loads movies..json instead: those maps cut to that language and the English the page falls back to, everything else as is. */ -}} {{- if eq $name "movies" -}} {{- range hugo.Sites -}} {{- $l := .Language.Lang -}} {{- $movies := slice -}} {{- range $d.movies -}} {{- $m := dict -}} {{- range $k, $v := . -}} {{- if in (slice "titles" "genres" "synopsis") $k -}} {{- $cut := dict -}} {{- range $vl, $vv := $v }}{{ if in (slice "en" $l) $vl }}{{ $cut = merge $cut (dict $vl $vv) }}{{ end }}{{ end -}} {{- $v = $cut -}} {{- end -}} {{- $m = merge $m (dict $k $v) -}} {{- end -}} {{- $movies = $movies | append $m -}} {{- end -}} {{- $slim := merge $d (dict "movies" $movies) -}} {{- with resources.FromString (printf "data/movies.%s.json" $l) (jsonify $slim) -}} {{- .Publish -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
