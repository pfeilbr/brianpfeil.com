{{- /* Post-processes rendered Markdown before it is printed: - task-list checkboxes (- [ ]) are decorative -- the item text is the label -- so hide them from assistive tech instead of announcing 40 unlabelled, disabled checkboxes; - raw HTML tags written straight into content (project pages) never pass through the image render hook, so give them the same lazy loading and async decoding it adds. */ -}} {{- $c := . -}} {{- $c = replace $c `` `` -}} {{- $c = replace $c `` `` -}} {{- /* Add to every , then drop the copy the render hook already put at the end of Markdown images (RE2 has no lookahead to skip them). */ -}} {{- $c = replaceRE `` `>` -}} {{- /* Raw tags pointing at dead remote images (see render-image.html): swap in the rescued copy, or replace the whole tag with the "no longer available" note. */ -}} {{- range $url, $local := hugo.Data.image_rewrites | default dict -}} {{- if in $c $url -}} {{- if $local -}} {{- $c = replace $c $url $local -}} {{- else -}} {{- $mark := printf "gone:%s" (md5 $url) -}} {{- $c = replace $c $url $mark -}} {{- $c = replaceRE (printf `]*%s[^>]*>` $mark) (printf `%s` (i18n "image_gone")) $c -}} {{- end -}} {{- end -}} {{- end -}} {{- return $c -}}
