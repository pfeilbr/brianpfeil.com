{{- /* Language switcher. Renders one link per configured language. A language the current page has no translation for links to that language's home page instead, and is marked so the click handler in baseof.html knows it is a fallback rather than a real translation. */ -}} {{- $cur := .Site.Language.Lang -}} {{- $translations := dict -}} {{- range .AllTranslations -}} {{- $translations = merge $translations (dict .Language.Lang .RelPermalink) -}} {{- end -}}
{{- range $i, $lang := hugo.Sites }} {{- $href := index $translations $lang.Language.Lang -}} {{- $isTranslated := ne $href nil -}} {{- /* defaultContentLanguageInSubdir is false, so the default language lives at the root. */ -}} {{- if not $isTranslated }}{{ $href = cond (eq $lang.Language.Lang hugo.Sites.Default.Language.Lang) "/" (printf "/%s/" $lang.Language.Lang) }}{{ end -}} {{- if $i }}·{{ end -}} {{ $lang.Language.Label }} {{- end }}
